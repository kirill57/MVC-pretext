[215-215 title appG-plot-sphere default_required]
Sphere

[217-221 p appG-plot-sphere default_required]
\mathbf r(\phi,\theta)
=
(R\sin\phi\cos\theta,\ R\sin\phi\sin\theta,\ R\cos\phi).

[223-233 program appG-plot-sphere default_required]
phi = np.linspace(0, np.pi, 150)
theta = np.linspace(0, 2*np.pi, 150)
Phi, Theta = np.meshgrid(phi, theta)

X = R*np.sin(Phi)*np.cos(Theta)
Y = R*np.sin(Phi)*np.sin(Theta)
Z = R*np.cos(Phi)

ax.plot_surface(X, Y, Z)
